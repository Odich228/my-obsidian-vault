#!/bin/bash
#
# Samba Metrics Collector for Node Exporter Textfile Collector
# Скрипт собирает метрики Samba и сохраняет их в формате Prometheus
#

# Конфигурация
METRIC_FILE="/var/lib/prometheus/node-exporter/samba.prom"
TEMP_FILE="${METRIC_FILE}.tmp.$$"  # $$ - PID процесса для уникальности
LOCK_FILE="/var/run/samba_metrics.lock"

# Функция для безопасного завершения
cleanup() {
    rm -f "$TEMP_FILE"
    rm -f "$LOCK_FILE"
    exit 1
}

trap cleanup INT TERM

# Проверяем, не запущен ли уже другой экземпляр
if [ -f "$LOCK_FILE" ]; then
    PID=$(cat "$LOCK_FILE" 2>/dev/null)
    if kill -0 "$PID" 2>/dev/null; then
        echo "Another instance is already running (PID: $PID)"
        exit 1
    fi
fi
echo $$ > "$LOCK_FILE"

# Создаем временный файл
mkdir -p "$(dirname "$METRIC_FILE")"

# Начинаем запись метрик
cat > "$TEMP_FILE" << 'HEADER'
# HELP samba_connections_total Total number of active Samba connections
# TYPE samba_connections_total gauge
# HELP samba_open_files_total Total number of open files in Samba
# TYPE samba_open_files_total gauge
# HELP samba_processes_count Number of running Samba processes (smbd/nmbd)
# TYPE samba_processes_count gauge
# HELP samba_info Samba server information
# TYPE samba_info gauge
# HELP samba_locked_files_total Number of locked files
# TYPE samba_locked_files_total gauge
# HELP samba_users_connected Number of unique connected users
# TYPE samba_users_connected gauge
HEADER

# Проверяем доступность smbstatus
if ! command -v smbstatus &> /dev/null; then
    echo "# ERROR: smbstatus command not found" >> "$TEMP_FILE"
    echo "samba_connections_total 0" >> "$TEMP_FILE"
    echo "samba_open_files_total 0" >> "$TEMP_FILE"
    echo "samba_processes_count 0" >> "$TEMP_FILE"
    echo "samba_info 0" >> "$TEMP_FILE"
    echo "samba_locked_files_total 0" >> "$TEMP_FILE"
    echo "samba_users_connected 0" >> "$TEMP_FILE"
    mv "$TEMP_FILE" "$METRIC_FILE"
    rm -f "$LOCK_FILE"
    exit 0
fi

# Получаем вывод smbstatus
SMBSTATUS=$(smbstatus 2>/dev/null)

# 1. Подсчитываем подключения (сессии)
# Формат: PID Username Group Machine
CONNECTIONS=$(echo "$SMBSTATUS" | \
    grep -E "^[[:space:]]*[0-9]+[[:space:]]+" | \
    grep -v "^[[:space:]]*PID" | \
    wc -l)

# 2. Подсчитываем открытые файлы
# Ищем секцию с открытыми файлами
OPEN_FILES=$(echo "$SMBSTATUS" | \
    sed -n '/^Locked files:/,/^[A-Z]/p' | \
    grep -E "^[[:space:]]*[0-9]+[[:space:]]+" | \
    wc -l)

# Если не найдено в Locked files, ищем в другой секции
if [ "$OPEN_FILES" -eq 0 ]; then
    OPEN_FILES=$(echo "$SMBSTATUS" | \
        grep -c "^[[:space:]]*[0-9]\+.*[ROWE]*" || echo 0)
fi

# 3. Количество процессов Samba
SMBD_COUNT=$(pgrep -c smbd 2>/dev/null | tr -d '[:space:]' || echo 0)
NMBD_COUNT=$(pgrep -c nmbd 2>/dev/null | tr -d '[:space:]' || echo 0)
WINBINDD_COUNT=$(pgrep -c winbindd 2>/dev/null | tr -d '[:space:]' || echo 0)

# Убеждаемся что переменные не пустые
SMBD_COUNT=${SMBD_COUNT:-0}
NMBD_COUNT=${NMBD_COUNT:-0}
WINBINDD_COUNT=${WINBINDD_COUNT:-0}

TOTAL_PROCESSES=$((SMBD_COUNT + NMBD_COUNT + WINBINDD_COUNT))
# 4. Уникальные пользователи
UNIQUE_USERS=$(echo "$SMBSTATUS" | \
    grep -E "^[[:space:]]*[0-9]+[[:space:]]+" | \
    awk '{print $2}' | \
    sort -u | \
    grep -v "^$" | \
    wc -l)

# 5. Заблокированные файлы
LOCKED_FILES=$(smbstatus --locks 2>/dev/null | \
    grep -E "^[[:space:]]*[0-9]+[[:space:]]+" | \
    wc -l)

# 6. Версия Samba и информация
SAMBA_VERSION=$(smbd --version 2>/dev/null | head -1 || echo "unknown")
# Очищаем версию для использования в лейбле
SAMBA_VERSION_CLEAN=$(echo "$SAMBA_VERSION" | tr -d ' "')

# Записываем метрики
cat >> "$TEMP_FILE" << EOF
samba_connections_total ${CONNECTIONS:-0}
samba_open_files_total ${OPEN_FILES:-0}
samba_processes_count ${TOTAL_PROCESSES}
samba_locked_files_total ${LOCKED_FILES:-0}
samba_users_connected ${UNIQUE_USERS:-0}
samba_info{version="${SAMBA_VERSION_CLEAN}"} 1
EOF

# Атомарно перемещаем файл (важно для корректной работы)
mv "$TEMP_FILE" "$METRIC_FILE"

# Очищаем lock файл
rm -f "$LOCK_FILE"

exit 0
